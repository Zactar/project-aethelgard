```javascript
// Cohesive Milestone Plugin: v0.6.0 (Black Bile Geo-Vents, Resonant Heavy Cleave Sparks & Crushing Vault Ceilings)
window.StudioEngine.registerModule("Milestone_v0_6_0", {
  init: function(state, engine) {
    const w = state.world;
    const p = state.player;
    w.geoVents = w.geoVents || [];
    w.tarPools = w.tarPools || [];
    w.vaultCeilings = w.vaultCeilings || [];
    w.visualFx = w.visualFx || [];
    p.chargeRatio = p.chargeRatio || 0;
    p.isCharging = false;

    // Spawn 2 Geo-Vents and 2 Crypt Vault Fault-Lines per room depth
    if (w.geoVents.length === 0) {
      w.geoVents.push({ id: "gv_1", x: 260, y: 220, r: 24, intact: true });
      w.geoVents.push({ id: "gv_2", x: 540, y: 390, r: 24, intact: true });
    }
    if (w.vaultCeilings.length === 0) {
      w.vaultCeilings.push({ id: "vc_1", x: 380, y: 240, r: 54, hp: 100, maxHp: 100, triggered: false, collapseTimer: 0, collapsed: false });
      w.vaultCeilings.push({ id: "vc_2", x: 440, y: 410, r: 54, hp: 100, maxHp: 100, triggered: false, collapseTimer: 0, collapsed: false });
    }

    engine.on("PLAYER_ATTACK", (atk) => {
      if (p.chargeRatio >= 0.7) {
        this.checkCeilingImpact(state, p.x, p.y, (p.weaponDamage || 18) * 1.5);
      }
    });
  },

  checkCeilingImpact: function(state, x, y, poiseDmg) {
    const w = state.world;
    w.vaultCeilings.forEach(vc => {
      if (!vc.triggered && !vc.collapsed && Math.hypot(vc.x - x, vc.y - y) <= vc.r + 20) {
        vc.hp -= poiseDmg;
        w.cameraShake = Math.max(w.cameraShake || 0, 5);
        if (vc.hp <= 0 && !vc.triggered) {
          vc.triggered = true;
          vc.collapseTimer = 0.45;
        }
      }
    });
  },

  update: function(dt, state, engine) {
    if (state.paused) return;
    const p = state.player;
    const w = state.world;
    const k = state.keys;

    // --- 1. Diegetic Weapon Charge Mechanic & Resonance Sparks ---
    if (k.jump && p.stamina > 15 && p.attackTimer <= 0) {
      p.isCharging = true;
      p.chargeRatio = Math.min(1.0, (p.chargeRatio || 0) + dt * 1.35);
      p.stamina = Math.max(0, p.stamina - dt * 12);
    } else {
      if (p.isCharging && p.chargeRatio > 0.25) {
        p.attackTimer = 0.28;
        p.stamRegenDelay = 1.0;
        engine.emit("PLAYER_ATTACK", { charge: p.chargeRatio, heavy: p.chargeRatio >= 0.7 });
      }
      p.isCharging = false;
      p.chargeRatio = Math.max(0, (p.chargeRatio || 0) - dt * 3.5);
    }

    const rad = p.facingAngle || 0;
    const reach = (p.weaponReach || 34) * (1.0 + p.chargeRatio * 0.4);
    const tipX = p.x + Math.cos(rad) * reach;
    const tipY = p.y + Math.sin(rad) * reach;

    if (p.chargeRatio >= 0.5) {
      if (Math.random() < 0.45) {
        w.visualFx.push({
          type: "lightning", x1: tipX, y1: tipY,
          x2: tipX + (Math.random() - 0.5) * 32, y2: tipY + 12 + Math.random() * 16,
          life: 0.12, color: Math.random() < 0.5 ? "#a855f7" : "#f59e0b"
        });
      }
      if (p.chargeRatio >= 0.95 && Math.random() < 0.35) {
        w.visualFx.push({
          type: "fracture", x: p.x + (Math.random() - 0.5) * 8, y: p.y + 10 + (Math.random() - 0.5) * 6,
          len: 12 + Math.random() * 14, angle: Math.random() * Math.PI * 2, life: 0.45
        });
      }
    }

    // --- 2. Interactive Geo-Vents (Breach & Tar Coating) ---
    const breachDist = 38;
    w.geoVents.forEach(gv => {
      if (!gv.intact) return;
      const distP = Math.hypot(p.x - gv.x, p.y - gv.y);
      const kick = k.action && distP <= breachDist;
      const cleaveHit = p.attackTimer > 0 && Math.hypot(tipX - gv.x, tipY - gv.y) <= gv.r + 10;
      if (kick || cleaveHit) {
        gv.intact = false;
        w.cameraShake = Math.max(w.cameraShake || 0, 7);
        w.tarPools.push({ x: gv.x, y: gv.y, r: 64, life: 12.0, ignited: false, burnTimer: 0 });
        for (let i = 0; i < 18; i++) {
          (w.particles || (w.particles = [])).push({
            x: gv.x, y: gv.y, vx: (Math.random() - 0.5) * 160, vy: (Math.random() - 0.5) * 160,
            r: 3 + Math.random() * 4, life: 0.6, maxLife: 0.6, color: "#1c1917"
          });
        }
      }
    });

    // Update Tar Pools & Burn Inferno
    for (let i = w.tarPools.length - 1; i >= 0; i--) {
      const tp = w.tarPools[i];
      tp.life -= dt;
      if (!tp.ignited) {
        // Ignite if near active brazier or player torch swing
        (w.braziers || []).forEach(bz => {
          if (bz.lit && Math.hypot(bz.x - tp.x, bz.y - tp.y) <= tp.r + 14) tp.ignited = true;
        });
        if (p.attackTimer > 0 && p.torchLevel > 20 && Math.hypot(tipX - tp.x, tipY - tp.y) <= tp.r) {
          tp.ignited = true;
        }
      } else {
        tp.burnTimer = (tp.burnTimer || 0) + dt;
        w.cameraShake = Math.max(w.cameraShake || 0, 1.2);
        if (Math.random() < 0.4) {
          (w.particles || (w.particles = [])).push({
            x: tp.x + (Math.random() - 0.5) * tp.r * 1.5, y: tp.y + (Math.random() - 0.5) * tp.r * 1.5,
            vx: (Math.random() - 0.5) * 30, vy: -40 - Math.random() * 40,
            r: 3 + Math.random() * 3, life: 0.5, maxLife: 0.5, color: Math.random() < 0.5 ? "#f59e0b" : "#dc2626"
          });
        }
      }

      // Entities in Tar: apply slow, bile coat, or fire DoT
      const targets = [p, ...(w.enemies || [])];
      targets.forEach(e => {
        if (!e || e.health <= 0) return;
        if (Math.hypot(e.x - tp.x, e.y - tp.y) <= tp.r) {
          e.speedMult = tp.ignited ? 0.85 : 0.55;
          e.bileCoated = 8.0;
          if (tp.ignited) {
            e.health -= 28 * dt;
            if (Math.random() < 0.15) {
              (w.floatingTexts || (w.floatingTexts = [])).push({
                text: "28 FIRE", x: e.x + (Math.random() - 0.5) * 16, y: e.y - 18, color: "#f59e0b", life: 0.6
              });
            }
          }
        }
      });
      if (tp.life <= 0) w.tarPools.splice(i, 1);
    }

    // --- 3. Unstable Crypt Vault Ceilings ---
    w.vaultCeilings.forEach(vc => {
      if (vc.collapsed) return;
      if (vc.triggered) {
        vc.collapseTimer -= dt;
        w.cameraShake = Math.max(w.cameraShake || 0, 4);
        if (vc.collapseTimer <= 0) {
          vc.collapsed = true;
          vc.triggered = false;
          w.cameraShake = 16;
          // Spawn permanent obstacle & rubble debris
          (w.obstacles || (w.obstacles = [])).push({ x: vc.x - 24, y: vc.y - 24, width: 48, height: 48, rubble: true });
          // Crush damage to all entities
          const affected = [p, ...(w.enemies || [])];
          affected.forEach(ent => {
            if (Math.hypot(ent.x - vc.x, ent.y - vc.y) <= vc.r) {
              ent.health -= 120;
              ent.stun = Math.max(ent.stun || 0, 1.8);
              (w.floatingTexts || (w.floatingTexts = [])).push({
                text: "120 CRUSH!", x: ent.x, y: ent.y - 26, color: "#eab308", life: 1.2
              });
            }
          });
        }
      }
    });

    // Clean up temporary visual effects
    for (let i = w.visualFx.length - 1; i >= 0; i--) {
      w.visualFx[i].life -= dt;
      if (w.visualFx[i].life <= 0) w.visualFx.splice(i, 1);
    }
  },

  render: function(ctx, state, engine) {
    if (state.paused) return;
    const w = state.world;
    const p = state.player;
    const lp = w.lowPoly;

    ctx.save();
    // Diegetic World Shake Context
    if (w.cameraShake > 0) {
      const sx = (Math.random() - 0.5) * w.cameraShake;
      const sy = (Math.random() - 0.5) * w.cameraShake;
      ctx.translate(sx, sy);
    }

    // 1. Render Tar Pools
    w.tarPools.forEach(tp => {
      ctx.save();
      ctx.beginPath();
      ctx.arc(tp.x, tp.y, tp.r, 0, Math.PI * 2);
      ctx.fillStyle = tp.ignited ? "rgba(185, 28, 28, 0.75)" : "rgba(15, 23, 42, 0.88)";
      ctx.fill();
      ctx.lineWidth = 2.5;
      ctx.strokeStyle = tp.ignited ? "#f59e0b" : "#475569";
      ctx.stroke();
      if (tp.ignited) {
        ctx.beginPath();
        ctx.arc(tp.x, tp.y, tp.r * 0.6, 0, Math.PI * 2);
        ctx.fillStyle = "rgba(234, 179, 8, 0.45)";
        ctx.fill();
      }
      ctx.restore();
    });

    // 2. Render Geo-Vents
    w.geoVents.forEach(gv => {
      if (lp && lp.drawExtrudedBox) {
        lp.drawExtrudedBox(ctx, gv.x - 14, gv.y - 14, 28, 28, gv.intact ? 14 : 4, gv.intact ? "#1e1b4b" : "#0f172a", "#312e81", p.x, p.y);
      }
      ctx.save();
      ctx.beginPath();
      ctx.arc(gv.x, gv.y, 8, 0, Math.PI * 2);
      ctx.fillStyle = gv.intact ? "#6366f1" : "#111827";
      ctx.fill();
      ctx.restore();
    });

    // 3. Render Vault Fault-Lines & Telegraphs
    w.vaultCeilings.forEach(vc => {
      if (vc.collapsed) {
        ctx.save();
        ctx.fillStyle = "#292524";
        ctx.beginPath();
        ctx.arc(vc.x, vc.y, 22, 0, Math.PI * 2);
        ctx.fill();
        ctx.strokeStyle = "#78716c";
        ctx.stroke();
        ctx.restore();
        return;
      }
      ctx.save();
      ctx.beginPath();
      ctx.arc(vc.x, vc.y, vc.r, 0, Math.PI * 2);
      ctx.setLineDash([4, 6]);
      ctx.lineWidth = 2;
      ctx.strokeStyle = vc.triggered ? "#dc2626" : "rgba(234, 179, 8, 0.4)";
      ctx.stroke();
      ctx.setLineDash([]);
      if (vc.triggered) {
        ctx.fillStyle = "rgba(220, 38, 38, 0.22)";
        ctx.fill();
      }
      ctx.restore();
    });

    // 4. Render Resonant Visual FX (Lightning, Fractures, Sparks)
    w.visualFx.forEach(fx => {
      ctx.save();
      if (fx.type === "lightning") {
        ctx.strokeStyle = fx.color;
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.moveTo(fx.x1, fx.y1);
        const midX = (fx.x1 + fx.x2) * 0.5 + (Math.random() - 0.5) * 8;
        const midY = (fx.y1 + fx.y2) * 0.5 + (Math.random() - 0.5) * 8;
        ctx.lineTo(midX, midY);
        ctx.lineTo(fx.x2, fx.y2);
        ctx.stroke();
      } else if (fx.type === "fracture") {
        ctx.strokeStyle = "#eab308";
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        ctx.moveTo(fx.x, fx.y);
        ctx.lineTo(fx.x + Math.cos(fx.angle) * fx.len, fx.y + Math.sin(fx.angle) * fx.len);
        ctx.stroke();
      }
      ctx.restore();
    });

    // 5. Heavy Charge Visual Aura & Articulated Resonating Sweep
    if (p.chargeRatio > 0.4) {
      ctx.save();
      ctx.beginPath();
      ctx.arc(p.x, p.y, 20 + p.chargeRatio * 12, 0, Math.PI * 2);
      ctx.strokeStyle = p.chargeRatio >= 0.95 ? "#eab308" : "#a855f7";
      ctx.lineWidth = 2 + p.chargeRatio * 2;
      ctx.globalAlpha = 0.45 + Math.sin(state.tick * 0.3) * 0.25;
      ctx.stroke();
      ctx.restore();
    }

    ctx.restore();
    ctx.setTransform(1, 0, 0, 1, 0, 0); // Steady Ribbon Pass
  }
});
```
